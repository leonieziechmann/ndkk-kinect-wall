Eigener Client in 10 Zeilen

import { KinectStream } from '/lib/kinect-stream.js';

const kinect = new KinectStream({ streams: ['depth', 'lut'] });
kinect.addEventListener('depth', (e) => {
  const { data, width, height, seq } = e.detail;   // Uint16Array, mm, 0 = kein Messwert
});
kinect.addEventListener('lut', (e) => {
  const rays = e.detail.data;                        // Float32Array x,y je Pixel: Punkt = (x*z, y*z, z)
});
kinect.connect();
